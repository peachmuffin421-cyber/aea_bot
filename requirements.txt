# ============================================================================
# ÆEA TELEGRAM BOT — PYTHON DEPENDENCY MANIFEST
# Version: 2.3.1 (MQL4 parity — Signal Firing & Extrema Fallback Patch)
# Author: Natalia Tanyatia
# Platform: Heroku + Telegram Bot + Local HTTPS Bridge (index.html)
# Python: >=3.10 (required for modern typing syntax and asyncio features)
# ============================================================================

# ----------------------------------------------------------------------------
# CORE TELEGRAM INTEGRATION
# ----------------------------------------------------------------------------
# python-telegram-bot v20+ uses the async/await API (Application, CommandHandler,
# CallbackQueryHandler). MUST be pinned to >=20.0,<22.0 for compatibility with
# the `Application.builder().token(...).post_init(...).build()` API used in app.py.
python-telegram-bot>=20.7,<22.0

# httpx is a transitive dependency of python-telegram-bot v20+, but pinning it
# ensures the async HTTP client used for Telegram API calls stays at a version
# compatible with the async polling loop.
httpx>=0.26.0,<0.28.0

# ----------------------------------------------------------------------------
# SYMBOLIC ARITHMETIC (OPTIONAL — core uses Python stdlib `fractions` + `decimal`)
# ----------------------------------------------------------------------------
# `sympy` is imported by the ÆI Codex (setup.sh) and by any symbolic-layer
# extensions that use exact rational quaternion arithmetic. The core app.py
# uses stdlib `Fraction`/`Decimal` only, but Heroku deployments that also run
# the GAIA/ÆI seed require sympy present.
sympy>=1.12,<2.0

# ----------------------------------------------------------------------------
# HTTP CLIENT (Worker Relay communication)
# ----------------------------------------------------------------------------
# `urllib.request` is used by app.py's FeedClient (stdlib, no external dep).
# However, `httpx` above provides async HTTP; `requests` is retained here as
# a fallback for any synchronous diagnostic scripts or Heroku scheduler jobs.
requests>=2.31.0,<3.0

# `certifi` is required by `requests` for TLS certificate verification against
# the Cloudflare Worker relay endpoint (https://*.workers.dev).
certifi>=2024.2.2

# ----------------------------------------------------------------------------
# TIMEZONE HANDLING (for diagnostic timestamp formatting)
# ----------------------------------------------------------------------------
# `pytz` provides the tz database used when rendering timestamps in the
# /diagnostics and /signals endpoints (ISO format with timezone offsets).
pytz>=2024.1

# ----------------------------------------------------------------------------
# ENVIRONMENT VARIABLE LOADING (local development convenience)
# ----------------------------------------------------------------------------
# `python-dotenv` allows app.py to read `RELAY_URL`, `RELAY_KEY`, and
# `TELEGRAM_BOT_TOKEN` from a local `.env` file. On Heroku these come from
# `heroku config:set` and this package is a no-op.
python-dotenv>=1.0.0,<2.0

# ----------------------------------------------------------------------------
# ASYNCIO ENHANCEMENTS (Heroku free/eco dyno compatibility)
# ----------------------------------------------------------------------------
# `uvloop` provides a faster event loop on Linux (Heroku runtime). It is
# optional — app.py falls back to the default asyncio event loop if uvloop
# import fails. Do not pin to platform-specific wheels for local dev on
# Windows/macOS; pip will simply skip if no wheel matches.
uvloop>=0.19.0; sys_platform == "linux"

# ----------------------------------------------------------------------------
# LOGGING & OBSERVABILITY (Heroku log drains)
# ----------------------------------------------------------------------------
# `colorlog` provides colorized log output during local development. On
# Heroku, structured plain-text logs are preferred, so this is optional.
colorlog>=6.8.0

# ----------------------------------------------------------------------------
# RUNTIME PATCHES FOR PYTHON TELEGRAM BOT ON HEROKU
# ----------------------------------------------------------------------------
# Ensures that asyncio tasks spawned by the Application polling loop are
# cancelled cleanly on SIGTERM (Heroku dyno cycling).
# No external package required — handled via atexit + signal in app.py.

# ============================================================================
# DEPLOYMENT NOTES
# ============================================================================
# 1. Heroku runtime uses the `heroku/python` buildpack.
# 2. Set `runtime.txt` to `python-3.11.8` (or >=3.10) for compatible wheels.
# 3. Heroku reads `Procfile`:
#       web: python app.py
#    (Heroku injects $PORT; app.py reads it via BRIDGE_PORT).
# 4. Required Heroku config vars:
#       TELEGRAM_BOT_TOKEN = <your_bot_token>
#       RELAY_URL          = https://<your-worker>.workers.dev
#       RELAY_KEY          = <optional_shared_secret>
#       PORT               = <injected by Heroku>
# 5. Optional config vars:
#       BRIDGE_TUNNEL      = cloudflared | ngrok | serveo
#       BRIDGE_USE_TLS     = 1 (with BRIDGE_TLS_CERT/KEY)
#       TELEGRAM_WEBAPP_URL= https://<public-tunnel-url>
#       CORS_ORIGIN        = *
#       UPDATE_INTERVAL    = 30
# ============================================================================
